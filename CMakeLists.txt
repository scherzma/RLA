cmake_minimum_required(VERSION 3.24)
project(RLA VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "RLA requires 64-bit Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Match the x64-windows-static vcpkg triplet. The release needs no VC runtime installer.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(imgui CONFIG REQUIRED)
find_package(implot CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_library(rla_core STATIC
    src/App.cpp src/Analyzer.cpp src/Latency.cpp src/MouseLibrary.cpp
    src/MouseLibraryUI.cpp src/Autosave.cpp src/DataStore.cpp src/DeviceManager.cpp
    src/InputEngine.cpp src/Renderer.cpp)
target_compile_definitions(rla_core PUBLIC UNICODE _UNICODE)
target_include_directories(rla_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(rla_core PUBLIC imgui::imgui implot::implot nlohmann_json::nlohmann_json
    d3d11 dxgi d3dcompiler comdlg32 shell32 user32 gdi32 imm32 dwmapi bcrypt)
if(MSVC)
    target_compile_options(rla_core PUBLIC /utf-8 /EHsc)
endif()
add_executable(RLA WIN32 main.cpp RLA.rc)
target_link_libraries(RLA PRIVATE rla_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(RLARegression tests/Regression.cpp)
    target_link_libraries(RLARegression PRIVATE rla_core)
    add_test(NAME regression COMMAND RLARegression)
    set_tests_properties(regression PROPERTIES TIMEOUT 180 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
install(TARGETS RLA RUNTIME DESTINATION .)
install(FILES docs/INSTALL.md DESTINATION .)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/THIRD-PARTY-NOTICES.txt" "RLA third-party notices\n\n")
foreach(package imgui implot nlohmann-json)
    file(READ "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/${package}/copyright" license_text)
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/THIRD-PARTY-NOTICES.txt" "${package}\n${license_text}\n\n")
endforeach()
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/THIRD-PARTY-NOTICES.txt" DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "RLA-${PROJECT_VERSION}-windows-x64")
include(CPack)
